# Приёмы TLS из экосистемы nfqws2-keenetic.
# Источники: github.com/nfqws/nfqws2-keenetic 1.3.1 (три стратегии
# circular из nfqws2.conf — здесь по отдельности, чтобы сканер мерил
# каждую) и чат t.me/nfqws, сентябрь–октябрь 2026: только nfqws2-приёмы
# с подтверждением «работает». Это «приёмы» — сканер сам оборачивает их
# под цель (--filter-tcp/--filter-l7/--payload/--hostlist).
# seqovl_pattern=<имя блоба> дозаявляет blob_registry (stun, stun2,
# tls_clienthello — синоним tls_google из nfqws2-keenetic).

[kn2_tls_fake_fonts_multisplit_seqovl]
name = nfqws2-keenetic №1: fake (SNI fonts.google.com) + multisplit seqovl=1
author = nfqws/nfqws2-keenetic
label = recommended
description = Стратегия 1 дефолта nfqws2-keenetic: fake ClientHello Google с SNI fonts.google.com и сдвигом seq, затем multisplit по 1,midsld с перекрытием в 1 байт из того же ClientHello.
--lua-desync=fake:blob=tls_clienthello:tls_mod=rnd,dupsid,sni=fonts.google.com:tcp_seq=10000
--lua-desync=multisplit:pos=1,midsld:seqovl=1:seqovl_pattern=tls_clienthello:tcp_ts_up

[kn2_tls_zero_fake_multisplit]
name = nfqws2-keenetic №2: нулевой fake + multisplit 1,midsld
author = nfqws/nfqws2-keenetic
label = stable
description = Стратегия 2 дефолта nfqws2-keenetic: 4 нулевых байта как fake (с tls_mod он превращается в ClientHello www.google.com) ×2 с плохим ack, затем multisplit по 1,midsld.
--lua-desync=fake:blob=0x00000000:tcp_ack=-66000:tls_mod=rnd,dupsid,sni=www.google.com:repeats=2
--lua-desync=multisplit:pos=1,midsld

[kn2_tls_hostfakesplit_ozon]
name = nfqws2-keenetic №3: hostfakesplit ozon.ru + seqovl
author = nfqws/nfqws2-keenetic
label = stable
description = Стратегия 3 дефолта nfqws2-keenetic: hostfakesplit с фейковым хостом ozon.ru, перекрытием до sniext+3 и md5/ts-фулингом, затем второй hostfakesplit.
--lua-desync=hostfakesplit:host=ozon.ru:midhost=host-2:seqovl=sniext+3:seqovl_pattern=tls_clienthello:badsum:tcp_md5:tcp_ts_up
--lua-desync=hostfakesplit:tcp_md5:tcp_ts_up

[kn2_tls_multisplit_seqovl568_stun]
name = 16 КБ-блок: multisplit 1,midsld, seqovl=568 из STUN
author = nik-ps (чат nfqws2-keenetic)
label = recommended

[kn2_tls_dot_multisplit_seqovl4]
name = multisplit 1,host+2,midsld+2,endsld-2 seqovl=4 + fake ozon.ru
author = Andrey R (чат nfqws2-keenetic)
label = stable
description = Из профиля для DoT/DoH: четыре разреза вокруг имени хоста с перекрытием 4 байта + fake tls_google с SNI ozon.ru, md5 и сдвигом seq ×3. Подтверждено в чате для публичных резолверов.
--lua-desync=multisplit:pos=1,host+2,midsld+2,endsld-2:seqovl=4
--lua-desync=fake:blob=tls_google:tcp_seq=-10000:tcp_md5:tls_mod=rnd,dupsid,sni=ozon.ru:repeats=3

[kn2_tls_yt_multisplit11_fake_rutube]
name = multisplit по 11 точкам seqovl=3 + fake rutube.ru
author = Andrey R (чат nfqws2-keenetic)
